function buildQuestion(raw){
  let opts = Array.isArray(raw.options) ? raw.options.map(String) : [String(raw.correct), '0', '1'];
  const correct = String(raw.correct);
  // make sure the correct answer is present exactly once and every option is unique
  const seen = new Set(); const out = [];
  opts.forEach(o => { if (!seen.has(o)) { seen.add(o); out.push(o); } });
  if (!seen.has(correct)) out.splice(getRandomInt(0, out.length), 0, correct);
  const numeric = out.every(o => /^[−-]?\d+(\.\d+)?$/.test(o));
  let guard = 0;
  while (out.length < 3 && guard++ < 20) {
    let cand;
    if (numeric) { const n = parseFloat(correct.replace('−', '-')) + (getRandomInt(1, 4) * (getRandomInt(0, 1) ? 1 : -1)); cand = n < 0 ? '−' + Math.abs(n) : String(n); }
    else cand = String(getRandomInt(2, 99));
    if (!seen.has(cand)) { seen.add(cand); out.splice(getRandomInt(0, out.length), 0, cand); }
  }
  while (out.length > 3) { const i = out.findIndex(o => o !== correct); out.splice(i, 1); }
  return { gr: raw.gr, en: raw.en, o: out, a: out.indexOf(correct) };
}
function generateQuestions(pool, count){
  const gens = questionTypes[pool], res = []; let last = -1;
  for (let i = 0; i < count; i++) {
    let gi, tries = 0;
    do { gi = getRandomInt(0, gens.length - 1); tries++; } while (gi === last && tries < 6 && gens.length > 1);
    last = gi;
    res.push(buildQuestion(gens[gi]()));
  }
  return res;
}

function renderStatsPanel(){
  const totals = safeGetJSON('mc_totals', { correct: 0, wrong: 0 }), topics = safeGetJSON('mc_topic_stats', {});
  const n = totals.correct + totals.wrong, acc = n ? Math.round(totals.correct / n * 100) : 0, games = parseInt(safeGet('mc_games_played')) || 0;
  const rows = Object.entries(topics).map(([k, s]) => { const tt = s.correct + s.wrong; return { k, tt, acc: tt ? Math.round(s.correct / tt * 100) : 0 }; })
    .filter(r => r.tt > 0).sort((a, b) => a.acc - b.acc);
  let h = `<div class="stat-big"><b>${acc}%</b><small>${tr('overall')} · ${games} ${tr('games')}</small></div>`;
  if (!rows.length) h += `<p style="text-align:center;opacity:.7">${tr('playMore')}</p>`;
  rows.forEach(r => {
    const col = r.acc >= 75 ? 'var(--success)' : r.acc >= 50 ? 'var(--gold)' : 'var(--danger)';
    h += `<div class="topic-row"><span class="nm">${topicName(r.k)}</span><span class="bar"><i style="width:${r.acc}%;background:${col}"></i></span><span class="pc" style="color:${col}">${r.acc}%</span></div>`;
  });
  $('statsBody').innerHTML = h;
}
function classifyTopic(q){
  const x = (q.en || '') + ' ' + (q.gr || '');
  if (/\[Complex|\[Μιγαδικοί|\|\d+ \+ \d+i\||\bi[²³⁴]/.test(x)) return 'Complex Numbers';
  if (/∫|′|\blim\b/.test(x)) return 'Calculus';
  if (/log|ln\(|ln /i.test(x)) return 'Logarithms';
  if (/sin\d|cos\d|tan\d|ημ\d|συν\d|εφ\d|sin²|ημ²|sin\(|ημ\(/i.test(x)) return 'Trigonometry';
  if (/C\(\d|τρόπους|ways|χειραψ|handshake|διαγών|diagonal|υποσύνολ|subset|!/.test(x)) return 'Combinatorics';
  if (/Πιθανότητα|Probability|P\(|Μέσος όρος|Average|Διάμεσος|Median/.test(x)) return 'Statistics & Probability';
  if (/πρόοδος|sequence|Σειρά|Series|1\+2\+3|Άθροισμα των 10/i.test(x)) return 'Sequences';
  if (/GCD|LCM|ΕΚΠ|ΜΚΔ|prime|πρώτ|mod |διαιρέτ|divisor|πολλαπλάσι|multiple|ψηφίο|last digit|άρτι|even/i.test(x)) return 'Number Theory';
  if (/Quadrant|Τεταρτημόριο|πR|Εμβαδόν|Area|Περίμετρος|Perimeter|γωνί|angle|πλευρ|sides|Κλίση|Slope|Απόσταση μεταξύ|Distance between|ακτίνας|radius|τρίγων|triangle/i.test(x)) return 'Geometry';
  if (/%/.test(x)) return 'Percentages';
  if (/x[²³\^]|\(x[+\-]|\dx\b|\bx =|ρίζ(ες|ων)|roots|Διακρίνουσα|Discriminant|det \[|x \+ 1\/x|a \+ b/.test(x)) return 'Algebra';
  if (/\d\/\d/.test(x)) return 'Fractions';
  if (/[²³]|√|\^|\d⁻/.test(x)) return 'Powers & Roots';
  return 'Arithmetic';
}
function recordTopicStat(q, ok){
  const topic = classifyTopic(q);
  (sessionTopicStats[topic] = sessionTopicStats[topic] || { correct: 0, wrong: 0 })[ok ? 'correct' : 'wrong']++;
  const life = safeGetJSON('mc_topic_stats', {});
  (life[topic] = life[topic] || { correct: 0, wrong: 0 })[ok ? 'correct' : 'wrong']++;
  safeSetJSON('mc_topic_stats', life);
  const tot = safeGetJSON('mc_totals', { correct: 0, wrong: 0 }); tot[ok ? 'correct' : 'wrong']++; safeSetJSON('mc_totals', tot);
}

function nextQuestion(){
  if (currentQuestion >= ACTIVE_QUESTIONS.length) {
    if (survivalMode) ACTIVE_QUESTIONS = ACTIVE_QUESTIONS.concat(generateQuestions(LEVEL_CONFIG[level].pool, 20));
    else { endGame(true); return; }
  }
  answered = false;
  if (!survivalMode) $('pill-' + currentQuestion)?.classList.add('current');
  $('answer-reveal')?.remove();

  let q = ACTIVE_QUESTIONS[currentQuestion];
  const ap = adaptivePool(), boosted = adaptiveEnabled(level) && ap !== LEVEL_CONFIG[level].pool;
  $('adaptiveBadge').classList.toggle('hidden', !boosted);
  if (boosted) q = generateQuestions(ap, 1)[0];
  isBossActive = level !== 'daily' && isBossIdx(currentQuestion);
  if (isBossActive) q = generateQuestions(POOL_ORDER[Math.min(POOL_ORDER.length - 1, basePoolIndex(level) + 1)], 1)[0];
  $('bossBanner').classList.toggle('hidden', !isBossActive);
  ACTIVE_QUESTIONS[currentQuestion] = q;

  const txt = lang === 'gr' ? q.gr : q.en, qEl = $('question');
  qEl.innerHTML = fmt(txt);
  qEl.className = 'question' + (txt.length > 58 ? ' xl' : txt.length > 26 ? ' long' : '');

  const box = $('options'); box.innerHTML = '';
  q.o.forEach((v, i) => {
    const b = document.createElement('button'); b.type = 'button'; b.className = 'option';
    b.innerHTML = `<kbd>${i + 1}</kbd>`; b.insertAdjacentHTML('beforeend', fmt(v));
    b.onclick = () => checkAnswer(i, b); box.appendChild(b);
  });
  const ch = $('chaserx'); ch.classList.remove('snap');
  startTimer();
}

function fmt(s){ return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/(\d|\))\^(\d+)/g,'$1<sup>$2</sup>'); }
function stopTimer(){ if (tickId) { clearInterval(tickId); tickId = null; } }
function startTimer(){
  stopTimer();
  qDur = LEVEL_CONFIG[level].baseTime * comboTimeFactor() * 1000;
  qEnd = performance.now() + qDur;
  $('bar').parentElement.classList.remove('warning', 'danger');
  tickId = setInterval(tick, 50); tick();
}
function tick(){
  const left = Math.max(0, qEnd - performance.now()), frac = left / qDur; lastFrac = frac;
  $('bar').style.width = (frac * 100) + '%';
  $('timerBar').setAttribute('aria-valuenow', Math.round(frac * 100));
  $('timeText').textContent = (left / 1000).toFixed(1) + (lang === 'gr' ? ' δλ' : ' s');
  const tb = $('timerBar');
  tb.classList.toggle('warning', frac < 0.3 && frac >= 0.15); tb.classList.toggle('danger', frac < 0.15);
  $('chaserx').style.left = (2 + (1 - frac) * 58) + '%';
  $('track').classList.toggle('danger', frac < 0.3);
  if (left <= 0) { stopTimer(); timeUp(); }
}
function pauseGame(on){
  if (gameOver || $('game').classList.contains('hidden')) return;
  if (on && !paused && !answered) { paused = true; pausedLeft = Math.max(0, qEnd - performance.now()); stopTimer(); $('pauseOv').classList.remove('hidden'); }
  else if (!on && paused) { paused = false; qEnd = performance.now() + pausedLeft; $('pauseOv').classList.add('hidden'); tickId = setInterval(tick, 50); }
}
document.addEventListener('visibilitychange', () => { if (document.hidden) pauseGame(true); });
document.addEventListener('keydown', e => {
  if ($('game').classList.contains('hidden') || e.ctrlKey || e.metaKey || e.altKey) return;
  if (e.key === 'p' || e.key === 'P' || e.key === 'Escape') { e.preventDefault(); pauseGame(!paused); return; }
  if (paused || answered) return;
  const n = parseInt(e.key);
  if (n >= 1 && n <= 3) { const b = $('options').children[n - 1]; if (b && !b.disabled) { e.preventDefault(); b.click(); } }
});

  const wrong = btns.map((b, i) => i).filter(i => i !== q.a && !btns[i].classList.contains('gone'));
  if (!wrong.length) return;
  const i = wrong[getRandomInt(0, wrong.length - 1)];
  btns[i].classList.add('gone'); btns[i].disabled = true;
  fiftyUsed = true; $('life5050').disabled = true; playSound('click');
}

}

function checkAnswer(idx, btn){
  if (answered || paused || gameOver) return;
  answered = true; stopTimer();
  const q = ACTIVE_QUESTIONS[currentQuestion], ok = idx === q.a;
  questionAnswers.push(ok); recordTopicStat(q, ok);
  if (ok) {
    btn.classList.add('correct'); playSound('correct');
    const boss = isBossActive; score += boss ? 2 : 1; if (boss) bossCorrect = true;
    combo++; maxCombo = Math.max(maxCombo, combo); speedSamples.push(lastFrac);
    const r = btn.getBoundingClientRect(), a = $('mc-card').getBoundingClientRect();
    createParticles(r.left + r.width / 2 - a.left, r.top + r.height / 2 - a.top, '#00A97A');
    if (combo >= 3) { playSound('combo'); showComboText(); }
    const cc = $('combo-counter'); if (combo >= 2) { cc.style.visibility = 'visible'; cc.classList.remove('pop'); void cc.offsetWidth; cc.classList.add('pop'); }
    updateScoreDisplay(); updateProgressPills(currentQuestion, true);
    [...$('options').children].forEach(b => b.disabled = true);
    const ch = $('chaserx'); ch.classList.add('snap'); ch.style.left = '2%';
    const rn = $('runner'); rn.classList.remove('bounce'); void rn.offsetWidth; rn.classList.add('bounce');
    setTimeout(() => { if (gameOver) return; currentQuestion++; nextQuestion(); }, 420);
  } else {
    btn.classList.add('wrong'); playSound('wrong'); combo = 0;
    if (navigator.vibrate) try { navigator.vibrate(120); } catch (e) {}
    updateProgressPills(currentQuestion, false); revealCorrect(q); showReveal(q);
    $('track').classList.add('caught');
    setTimeout(() => endGame(false), 1900);
  }
}
function timeUp(){
  if (answered || gameOver) return;
  answered = true;
  const q = ACTIVE_QUESTIONS[currentQuestion];
  questionAnswers.push(false); recordTopicStat(q, false); combo = 0;
  playSound('wrong'); updateProgressPills(currentQuestion, false); revealCorrect(q); showReveal(q, true);
  $('track').classList.add('caught');
  setTimeout(() => endGame(false), 1900);
}

function endGame(won){
  if (gameOver) return;
  gameOver = true; stopTimer();
  const n = questionAnswers.length, okN = questionAnswers.filter(Boolean).length, acc = n ? Math.round(okN / n * 100) : 0;
  const streak = touchStreak();
  showScreen('result');
  $('resultMsg').textContent = won ? tr('escaped') : tr('caught');
  const ratio = survivalMode ? Math.min(1, score / 30) : okN / DECK_SIZE;
  const stars = won ? 3 : ratio >= 0.6 ? 2 : ratio >= 0.25 ? 1 : 0;
  $('stars').innerHTML = [0, 1, 2].map(i => `<s class="${i < stars ? 'on' : ''}" style="animation-delay:${i * 0.18}s">★</s>`).join('');
  $('rScore').textContent = score; $('rCombo').textContent = maxCombo; $('rAcc').textContent = acc + '%';

  if (level === 'daily') safeSetJSON('mc_daily_done', { date: dailyDateStr, score });
  checkAchievements(won, streak);
  const isRecord = saveAndShowBestScore();
  $('recBox').innerHTML = isRecord && score > 0 ? `<span class="newrec">${tr('newRecord')}</span>` : '';
  updateLeaderboard(); showSessionTip();
  lastResult = { won, score, combo: maxCombo, acc, lvl: level === 'daily' ? tr('daily') : STR[lang][level], surv: survivalMode };
  $('share-text').textContent = tr('share');
  playSound(won ? 'win' : 'lose'); if (won) confetti();
  try { ROOT.scrollIntoView({ behavior: 'smooth', block: 'nearest' }); } catch (e) {}
}
function showSessionTip(){
  const tip = $('sessionTip'), ent = Object.entries(sessionTopicStats);
  if (!ent.length) { tip.textContent = ''; return; }
  let worst = null; ent.forEach(([t, s]) => { if (s.wrong > 0 && (!worst || s.wrong > worst[1].wrong)) worst = [t, s]; });
  tip.textContent = worst ? tr('mistakes', worst[1].wrong, topicName(worst[0])) : tr('cleanRun');
}
function checkAchievements(won, streak){
  newAch = []; const add = k => { if (unlock(k)) newAch.push(k); };
  if (won) add('firstWin');
  if (questionAnswers.length > 0 && questionAnswers.every(Boolean)) add('perfectRun');
  if (maxCombo >= 5) add('comboMaster');
  if ((level === 'expert' || level === 'impossible') && won) add('mathGenius');
  const avg = speedSamples.length ? speedSamples.reduce((s, v) => s + v, 0) / speedSamples.length : 0;
  if (won && avg >= 0.6) add('speedDemon');
  if ((parseInt(safeGet('mc_games_played')) || 0) >= 5) add('persistent');
  if (bossCorrect) add('bossSlayer');
  if (survivalMode && score >= 25) add('survivor');
  if (level === 'daily' && won) add('dailyHero');
  if (streak >= 3) add('streak3');
  const tot = safeGetJSON('mc_totals', { correct: 0, wrong: 0 }); if (tot.correct >= 100) add('hundred');
  const box = $('newAchievements'), list = $('result-achievements'); list.innerHTML = '';
  box.classList.toggle('hidden', !newAch.length);
  newAch.forEach(k => {
    const b = document.createElement('span'); b.className = 'badge unlocked'; b.textContent = ACH[k].icon; b.title = achLabel(k);
    list.appendChild(b);
    const cap = document.createElement('div'); cap.style.cssText = 'width:100%;font-size:.78rem;font-weight:600;opacity:.85'; cap.textContent = achLabel(k);
    list.appendChild(cap);
  });
  if (newAch.length) playSound('combo');
}
function saveAndShowBestScore(){
  const key = level === 'daily' ? `mathchaser_best_daily_${dailyDateStr}` : bestKey(level, survivalMode);
  let best = parseInt(safeGet(key)) || 0, rec = false;
  if (score > best) { best = score; rec = true; safeSet(key, best); saveToLeaderboard(score); }
  submitToGlobalLeaderboard(score, level, survivalMode, level === 'daily', dailyDateStr);
  $('bestScore').textContent = `${tr('best')}: ${best}`;
  return rec;
}
function saveToLeaderboard(s){
  const suffix = level === 'daily' ? `_daily_${dailyDateStr}` : (survivalMode ? '_survival' : '');
  const k = `mc_leaderboard_${level}${suffix}`, lb = safeGetJSON(k, []);
  lb.push({ score: s, date: new Date().toISOString(), combo: maxCombo });
  lb.sort((a, b) => b.score - a.score); lb.splice(10); safeSetJSON(k, lb);
}
function updateLeaderboard(){
  const box = $('leaderboard-preview'); if (_globalView) return; box.innerHTML = '';
  TIERS.forEach(l => {
    const lb = safeGetJSON(`mc_leaderboard_${l}`, []), lbs = safeGetJSON(`mc_leaderboard_${l}_survival`, []);
    [[lb, ''], [lbs, ' ♾']].forEach(([arr, tag]) => {
      if (!arr.length) return;
      const e = document.createElement('div'); e.className = 'lb-entry';
      e.innerHTML = `<span>${LEVEL_CONFIG[l].ico} ${STR[lang][l]}${tag}</span><b>${arr[0].score} ${tr('pts')} <small style="font-weight:500;opacity:.6">· 🔥${arr[0].combo || 0}</small></b>`;
      box.appendChild(e);
    });
  });
  if (!box.children.length) box.innerHTML = `<p class="lb-empty">${tr('noScores')}</p>`;
}

async function shareResult(){
  if (!lastResult) return;
  const text = tr('shareText', lastResult), btn = $('share-text');
  try {
    if (navigator.share) { await navigator.share({ title: 'Math Chaser PRO', text, url: SITE_URL }); return; }
  } catch (e) { if (e && e.name === 'AbortError') return; }
  try { await navigator.clipboard.writeText(text); btn.textContent = tr('copied'); }
  catch (e) {
    const ta = document.createElement('textarea'); ta.value = text; ta.style.cssText = 'position:fixed;opacity:0'; document.body.appendChild(ta); ta.select();
    try { document.execCommand('copy'); btn.textContent = tr('copied'); } catch (e2) {} ta.remove();
  }
  setTimeout(() => { btn.textContent = tr('share'); }, 2000);
}

async function submitToGlobalLeaderboard(finalScore, lvl, surv, isDaily, dKey){
  if (!_dbNS) return;
  try {
    const uid = _myUid || 'anon'; let name = tr('player');
    try { if (_userNS && _userNS.profiles) { const ps = await _userNS.profiles([uid]); if (ps && ps[uid] && ps[uid].name) name = ps[uid].name; } } catch (e) {}
    if (isDaily) { await _dbNS.doc('mc_lb_daily/' + dKey + '_' + uid).set({ date: dKey, name, score: finalScore, updatedAt: Date.now() }); return; }
    const ref = _dbNS.doc((surv ? 'mc_lb_survival' : 'mc_lb_alltime') + '/' + uid + (surv ? '' : '_' + lvl));
    let best = 0; try { const s = await ref.get(); best = (s && s.data ? s.data().score : s && s.score) || 0; } catch (e) {}
    if (finalScore > best) await ref.set({ name, level: lvl, score: finalScore, updatedAt: Date.now() });
  } catch (e) {}
}
async function toggleGlobalLeaderboard(){
  _globalView = !_globalView;
  $('globalLbToggle').classList.toggle('active', _globalView); $('localLbToggle').classList.toggle('active', !_globalView);
  if (_globalView) await renderGlobalLeaderboard(); else updateLeaderboard();
}
async function renderGlobalLeaderboard(){
  const box = $('leaderboard-preview'); box.innerHTML = `<p class="lb-empty">${tr('loading')}</p>`;
  if (!_dbNS) { box.innerHTML = `<p class="lb-empty">${tr('unavailable')}</p>`; return; }
  try {
    let q = _dbNS.collection('mc_lb_alltime'); if (q.orderBy) q = q.orderBy('score', 'desc'); if (q.limit) q = q.limit(10);
    const res = await q.get(), docs = (res && res.docs) ? res.docs.map(d => d.data ? d.data() : d) : (Array.isArray(res) ? res : []);
    if (!docs.length) { box.innerHTML = `<p class="lb-empty">${tr('noGlobal')}</p>`; return; }
    box.innerHTML = '';
    docs.forEach((d, i) => { const e = document.createElement('div'); e.className = 'lb-entry'; const n = document.createElement('span'); n.textContent = `#${i + 1} ${d.name || tr('player')}`; const s = document.createElement('b'); s.textContent = `${d.score || 0} ${tr('pts')}`; e.append(n, s); box.appendChild(e); });
  } catch (e) { box.innerHTML = `<p class="lb-empty">${tr('failed')}</p>`; }
}
